class Account { private int accountNumber; private double balance; static int count = 0; Account(int accountNumber, double balance) { this.accountNumber = accountNumber; this.balance = balance; count++; } void display() { System.out.println("Account: " + accountNumber); System.out.println("Balance: " + balance); } } public class Main { public static void main(String[] args) { Account a1 = new Account(101, 5000); Account a2 = new Account(102, 8000); a1.display(); a2.display(); System.out.println("Total accounts: " + Account.count); } }
